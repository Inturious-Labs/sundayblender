{{- /* Renders every markdown image. Accepts the bundle image as `./pic.jpg` or `pic.jpg` and emits an absolute URL either way, so the same markup works on the site, on localhost, and inside the newsletter email. Same hook as the Digital Sovereignty Chronicle repo - keep the two in sync. */ -}} {{- $dst := .Destination -}} {{- $src := $dst -}} {{- $u := urls.Parse $dst -}} {{- if not $u.IsAbs -}} {{- with .Page.Resources.Get (strings.TrimPrefix "./" $u.Path) -}} {{- $src = .Permalink -}} {{- else -}} {{- $src = absURL $dst -}} {{- end -}} {{- end -}}
